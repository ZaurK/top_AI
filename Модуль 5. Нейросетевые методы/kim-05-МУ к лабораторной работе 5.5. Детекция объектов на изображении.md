### 2. Теоретические сведения

#### 2.1. Задача детекции объектов

Детекция объектов — задача компьютерного зрения, заключающаяся в определении наличия объектов на изображении и их локализации с помощью ограничивающих рамок (bounding boxes).

**Формат аннотации**:
- Координаты центра \( (cx, cy) \), ширина \( w \), высота \( h \);
- Или координаты углов \( (x_1, y_1, x_2, y_2) \).

#### 2.2. Архитектуры детекторов в PyTorch

**Faster R-CNN** (встроен в torchvision):
- Двухэтапный подход:
  1. **Region Proposal Network (RPN)** — предлагает регионы с объектами;
  2. **Классификатор** — классифицирует регионы и уточняет координаты;
- Высокая точность, но медленная скорость.

**SSD (Single Shot Detector)**:
- Одновременное предсказание на нескольких масштабах;
- Использует якорные прямоугольники (anchor boxes);
- Хороший баланс между скоростью и точностью.

**YOLO (You Only Look Once)**:
- Единая свёрточная сеть, предсказывающая bounding boxes и классы;
- Разбивает изображение на сетку \( S \times S \);
- Очень высокая скорость работы (до 155 FPS).

#### 2.3. Метрики качества детекции

**IoU (Intersection over Union)**:
\[
IoU = \frac{Area(B_{pred} \cap B_{gt})}{Area(B_{pred} \cup B_{gt})}
\]

**AP (Average Precision)**:
- Площадь под кривой Precision-Recall.

**mAP (mean Average Precision)**:
- Среднее AP по всем классам.

   ```python
   class DetectionDataset(Dataset):
       def __init__(self, image_paths, annotations, transforms=None):
           """
           Args:
               image_paths: список путей к изображениям
               annotations: список словарей с ключами 'boxes' и 'labels'
               transforms: трансформации для изображений
           """
           self.image_paths = image_paths
           self.annotations = annotations
           self.transforms = transforms
       
       def __len__(self):
           return len(self.image_paths)
       
       def __getitem__(self, idx):
           # Загрузка изображения
           image = Image.open(self.image_paths[idx]).convert('RGB')
           
           # Получение аннотаций
           boxes = torch.as_tensor(self.annotations[idx]['boxes'], dtype=torch.float32)
           labels = torch.as_tensor(self.annotations[idx]['labels'], dtype=torch.int64)
           
           target = {
               'boxes': boxes,
               'labels': labels,
               'image_id': torch.tensor([idx])
           }
           
           if self.transforms:
               image = self.transforms(image)
           
           return image, target
   ```

   ```python
   def convert_to_coco_format(dataset, output_file):
       """Конвертация датасета в формат COCO"""
       coco_data = {
           'images': [],
           'annotations': [],
           'categories': []
       }
       
       # Добавление категорий
       categories = list(set([label for ann in dataset for label in ann['labels']]))
       for i, cat in enumerate(categories):
           coco_data['categories'].append({
               'id': i,
               'name': str(cat)
           })
       
       # Добавление изображений и аннотаций
       for idx, ann in enumerate(dataset):
           img = cv2.imread(ann['image_path'])
           h, w = img.shape[:2]
           
           coco_data['images'].append({
               'id': idx,
               'width': w,
               'height': h,
               'file_name': os.path.basename(ann['image_path'])
           })
           
           for bbox, label in zip(ann['boxes'], ann['labels']):
               x1, y1, x2, y2 = bbox
               coco_data['annotations'].append({
                   'id': len(coco_data['annotations']),
                   'image_id': idx,
                   'category_id': categories.index(label),
                   'bbox': [x1, y1, x2 - x1, y2 - y1],
                   'area': (x2 - x1) * (y2 - y1),
                   'iscrowd': 0
               })
       
       with open(output_file, 'w') as f:
           json.dump(coco_data, f)
       
       return coco_data
   ```

```python
def detect_with_fasterrcnn(model, image_path, confidence_threshold=0.5, device='cpu'):
    """
    Детекция объектов с помощью Faster R-CNN
    
    Args:
        model: загруженная модель
        image_path: путь к изображению
        confidence_threshold: порог уверенности
        device: устройство (cpu/cuda)
    
    Returns:
        boxes: список bounding boxes
        labels: список меток классов
        scores: список уверенностей
    """
    # Загрузка и предобработка изображения
    image = Image.open(image_path).convert('RGB')
    transform = transforms.ToTensor()
    image_tensor = transform(image).unsqueeze(0).to(device)
    
    # Инференс
    with torch.no_grad():
        predictions = model(image_tensor)
    
    # Обработка результатов
    boxes = predictions[0]['boxes'].cpu().numpy()
    scores = predictions[0]['scores'].cpu().numpy()
    labels = predictions[0]['labels'].cpu().numpy()
    
    # Фильтрация по порогу уверенности
    mask = scores >= confidence_threshold
    boxes = boxes[mask].astype(int)
    scores = scores[mask]
    labels = labels[mask]
    
    # Получение названий классов
    label_names = [COCO_CLASSES[int(l)] for l in labels]
    
    return boxes, label_names, scores
```

```python
def visualize_detection(image_path, boxes, labels, scores):
    """
    Визуализация результатов детекции
    
    Args:
        image_path: путь к изображению
        boxes: список bounding boxes [[x1, y1, x2, y2], ...]
        labels: список названий классов
        scores: список уверенностей
    """
    image = cv2.imread(image_path)
    image = cv2.cvtColor(image, cv2.COLOR_BGR2RGB)
    
    # Генерация цветов для классов
    unique_labels = list(set(labels))
    colors = np.random.randint(0, 255, (len(unique_labels), 3))
    label_to_color = {label: colors[i] for i, label in enumerate(unique_labels)}
    
    for box, label, score in zip(boxes, labels, scores):
        x1, y1, x2, y2 = box
        color = label_to_color[label]
        
        # Рисование прямоугольника
        cv2.rectangle(image, (x1, y1), (x2, y2), color.tolist(), 2)

                   cv2.FONT_HERSHEY_SIMPLEX, 0.6, color.tolist(), 2)
    
    plt.figure(figsize=(12, 8))
    plt.imshow(image)
    plt.axis('off')
    plt.tight_layout()
    plt.show()
```

```python
def get_fasterrcnn_model(num_classes):
    """
    Создание модели Faster R-CNN для кастомного количества классов
    
    Args:
        num_classes: количество классов (включая фон)
    
    Returns:
        model: модель Faster R-CNN
    """
    # Загрузка предобученной модели
    model = fasterrcnn_resnet50_fpn(pretrained=True)
    
    # Замена головы классификатора
    in_features = model.roi_heads.box_predictor.cls_score.in_features
    model.roi_heads.box_predictor = FastRCNNPredictor(in_features, num_classes)
    
    return model

def train_detection_model(model, train_loader, val_loader, num_epochs, device):
    """
    Обучение модели детекции
    
    Args:
        model: модель для обучения
        train_loader: DataLoader для обучения
        val_loader: DataLoader для валидации
        num_epochs: количество эпох
        device: устройство
    """
    # Оптимизатор
    params = [p for p in model.parameters() if p.requires_grad]
    optimizer = torch.optim.SGD(params, lr=0.005, momentum=0.9, weight_decay=0.0005)
    
    # Планировщик скорости обучения
    lr_scheduler = torch.optim.lr_scheduler.StepLR(optimizer, step_size=3, gamma=0.1)
    
    model.train()
    
    for epoch in range(num_epochs):
        epoch_loss = 0.0
        
        for images, targets in train_loader:
            images = [image.to(device) for image in images]
            targets = [{k: v.to(device) for k, v in t.items()} for t in targets]
            
            # Forward pass
            loss_dict = model(images, targets)
            losses = sum(loss for loss in loss_dict.values())
            
            # Backward pass
            optimizer.zero_grad()
            losses.backward()
            optimizer.step()
            
            epoch_loss += losses.item()
        
        lr_scheduler.step()
        
        # Валидация
        val_loss = validate_detection_model(model, val_loader, device)
        
        print(f'Epoch [{epoch+1}/{num_epochs}]')
        print(f'Train Loss: {epoch_loss/len(train_loader):.4f}')
        print(f'Val Loss: {val_loss:.4f}')
        print('-' * 40)
    
    return model

def validate_detection_model(model, val_loader, device):
    """
    Валидация модели детекции
    
    Args:
        model: модель
        val_loader: DataLoader для валидации
        device: устройство
    
    Returns:
        float: средняя потеря на валидации
    """
    model.eval()
    total_loss = 0.0
    
    with torch.no_grad():
        for images, targets in val_loader:
            images = [image.to(device) for image in images]
            targets = [{k: v.to(device) for k, v in t.items()} for t in targets]
            
            loss_dict = model(images, targets)
            losses = sum(loss for loss in loss_dict.values())
            total_loss += losses.item()
    
    model.train()
    return total_loss / len(val_loader)

# Подготовка датасета
def prepare_detection_data(annotations, image_dir, batch_size=4):
    """
    Подготовка данных для обучения детектора
    
    Args:
        annotations: список аннотаций
        image_dir: директория с изображениями
        batch_size: размер батча
    
    Returns:
        train_loader, val_loader: DataLoader'ы для обучения и валидации
    """
    # Разделение на train/val
    train_ann, val_ann = train_test_split(annotations, test_size=0.2, random_state=42)
    
    # Трансформации
    transform = transforms.Compose([
        transforms.ToTensor(),
    ])
    
    # Создание датасетов
    train_dataset = DetectionDataset(
        [ann['image_path'] for ann in train_ann],
        train_ann,
        transforms=transform
    )
    val_dataset = DetectionDataset(
        [ann['image_path'] for ann in val_ann],
        val_ann,
        transforms=transform
    )
    
    # Создание DataLoader'ов
    train_loader = DataLoader(train_dataset, batch_size=batch_size, 
                              shuffle=True, collate_fn=lambda x: tuple(zip(*x)))
    val_loader = DataLoader(val_dataset, batch_size=batch_size, 
                            shuffle=False, collate_fn=lambda x: tuple(zip(*x)))
    
    return train_loader, val_loader
```

```python
def analyze_parameters_fasterrcnn(model, image_path, confidence_range, device):
    """
    Анализ влияния порога уверенности на результаты детекции
    
    Args:
        model: модель детекции
        image_path: путь к изображению
        confidence_range: список порогов уверенности
        device: устройство
    
    Returns:
        list: результаты для каждого порога
    """
    results = []
    
    for conf in confidence_range:
        boxes, labels, scores = detect_with_fasterrcnn(
            model, image_path, confidence_threshold=conf, device=device
        )
        results.append({
            'confidence': conf,
            'num_objects': len(boxes),
            'avg_score': np.mean(scores) if len(scores) > 0 else 0
        })
    
    return results

def plot_parameter_analysis(results):
    """
    Визуализация влияния параметров
    """
    conf_values = [r['confidence'] for r in results]
    num_objects = [r['num_objects'] for r in results]
    avg_scores = [r['avg_score'] for r in results]
    
    fig, (ax1, ax2) = plt.subplots(1, 2, figsize=(12, 5))
    
    # Количество объектов
    ax1.plot(conf_values, num_objects, marker='o', linewidth=2)
    ax1.set_xlabel('Confidence Threshold')
    ax1.set_ylabel('Number of Detected Objects')
    ax1.set_title('Objects vs Confidence Threshold')
    ax1.grid(True, alpha=0.3)
    
    # Средняя уверенность
    ax2.plot(conf_values, avg_scores, marker='s', color='green', linewidth=2)
    ax2.set_xlabel('Confidence Threshold')
    ax2.set_ylabel('Average Confidence Score')
    ax2.set_title('Average Score vs Confidence Threshold')
    ax2.grid(True, alpha=0.3)
    
    plt.tight_layout()
    plt.show()
```

```python
def detect_with_yolo(model, image_path, confidence=0.5, iou=0.45):

    Детекция с использованием YOLO
    
    Args:
        model: модель YOLO

    Returns:
        boxes: список bounding boxes
        labels: список названий классов
        scores: список уверенностей

    labels = []
    scores = []
    
    for r in results:
        for box in r.boxes:
            x1, y1, x2, y2 = box.xyxy[0].tolist()
            conf = box.conf[0].tolist()
            cls = int(box.cls[0].tolist())
            
            boxes.append([int(x1), int(y1), int(x2), int(y2)])
            labels.append(model.names[cls])
            scores.append(conf)
    
    return boxes, labels, scores
```

```python
def prepare_yolo_dataset(annotations, output_dir='yolo_dataset'):
    """
    Подготовка датасета в формате YOLO
    
    Args:
        annotations: список аннотаций
        output_dir: директория для сохранения
    """
    os.makedirs(output_dir, exist_ok=True)
    os.makedirs(f'{output_dir}/images/train', exist_ok=True)
    os.makedirs(f'{output_dir}/labels/train', exist_ok=True)
    os.makedirs(f'{output_dir}/images/val', exist_ok=True)
    os.makedirs(f'{output_dir}/labels/val', exist_ok=True)
    
    # Создание файла с классами
    classes = list(set([label for ann in annotations for label in ann['labels']]))
    with open(f'{output_dir}/classes.txt', 'w') as f:
        for cls in classes:
            f.write(f'{cls}\n')
    
    # Разделение на train/val
    train_ann, val_ann = train_test_split(annotations, test_size=0.2, random_state=42)
    
    # Функция для конвертации в YOLO формат
    def yolo_format(bbox, img_width, img_height):
        x1, y1, x2, y2 = bbox
        x_center = (x1 + x2) / 2 / img_width
        y_center = (y1 + y2) / 2 / img_height
        width = (x2 - x1) / img_width
        height = (y2 - y1) / img_height
        return [x_center, y_center, width, height]
    
    # Создание аннотаций
    for ann in train_ann:
        img = cv2.imread(ann['image_path'])
        h, w = img.shape[:2]
        
        # Копирование изображения
        img_name = os.path.basename(ann['image_path'])
        shutil.copy(ann['image_path'], f'{output_dir}/images/train/{img_name}')
        
        # Создание файла с аннотацией
        label_name = img_name.replace('.jpg', '.txt').replace('.png', '.txt')
        with open(f'{output_dir}/labels/train/{label_name}', 'w') as f:
            for bbox, label in zip(ann['boxes'], ann['labels']):
                yolo_bbox = yolo_format(bbox, w, h)
                cls_id = classes.index(label)
                f.write(f"{cls_id} {yolo_bbox[0]} {yolo_bbox[1]} {yolo_bbox[2]} {yolo_bbox[3]}\n")
    
    # Аналогично для валидации
    for ann in val_ann:
        img = cv2.imread(ann['image_path'])
        h, w = img.shape[:2]
        
        img_name = os.path.basename(ann['image_path'])
        shutil.copy(ann['image_path'], f'{output_dir}/images/val/{img_name}')
        
        label_name = img_name.replace('.jpg', '.txt').replace('.png', '.txt')
        with open(f'{output_dir}/labels/val/{label_name}', 'w') as f:
            for bbox, label in zip(ann['boxes'], ann['labels']):
                yolo_bbox = yolo_format(bbox, w, h)
                cls_id = classes.index(label)
                f.write(f"{cls_id} {yolo_bbox[0]} {yolo_bbox[1]} {yolo_bbox[2]} {yolo_bbox[3]}\n")
    
    # Создание data.yaml
    with open(f'{output_dir}/data.yaml', 'w') as f:
        f.write(f"train: {output_dir}/images/train\n")
        f.write(f"val: {output_dir}/images/val\n")
        f.write(f"nc: {len(classes)}\n")
        f.write(f"names: {classes}\n")
    
    return classes

# Обучение YOLO
def train_yolo_model(data_yaml, epochs=50, imgsz=640, batch=16):
    """
    Обучение модели YOLO
    
    Args:
        data_yaml: путь к data.yaml
        epochs: количество эпох
        imgsz: размер изображений
        batch: размер батча
    
    Returns:
        model: обученная модель
    """
    from ultralytics import YOLO
    
    # Загрузка предобученной модели
    model = YOLO('yolov8n.pt')
    
    # Обучение
    results = model.train(
        data=data_yaml,
        epochs=epochs,
        imgsz=imgsz,
        batch=batch,
        device=0 if torch.cuda.is_available() else 'cpu',
        lr0=0.01,
        augment=True,
        patience=10
    )
    
    # Оценка модели
    metrics = model.val()
    print(f"mAP50-95: {metrics.box.map:.4f}")
    
    return model
```

```python
def compare_detectors(image_path, detectors, confidence_threshold=0.5):
    """
    Сравнение нескольких детекторов на одном изображении
    
    Args:
        image_path: путь к изображению
        detectors: словарь {name: (model, detect_function)}
        confidence_threshold: порог уверенности
    """

    if len(detectors) == 1:
        axes = [axes]

        # Визуализация

        axes[idx].set_title(f'{name}\nObjects: {len(boxes)}')
        axes[idx].axis('off')
```

#### 3.10. Пакетная обработка изображений

```python
def batch_detect(model, detect_func, image_dir, output_dir, confidence_threshold=0.5):
    """
    Пакетная детекция объектов
    
    Args:
        model: модель детекции
        detect_func: функция детекции
        image_dir: директория с изображениями
        output_dir: директория для сохранения результатов
        confidence_threshold: порог уверенности
    
    Returns:
        list: результаты для каждого изображения
    """
    os.makedirs(output_dir, exist_ok=True)
    results = []
    
    for img_file in os.listdir(image_dir):
        if img_file.lower().endswith(('.jpg', '.jpeg', '.png')):
            img_path = os.path.join(image_dir, img_file)
            
            # Детекция
            boxes, labels, scores = detect_func(model, img_path, confidence_threshold)
            
            # Сохранение результатов
            result = {
                'image': img_file,
                'num_objects': len(boxes),
                'labels': labels,
                'scores': scores,
                'boxes': boxes.tolist() if isinstance(boxes, np.ndarray) else boxes
            }
            results.append(result)
            
            # Визуализация и сохранение
            visualize_detection(img_path, boxes, labels, scores)
            plt.savefig(os.path.join(output_dir, f'result_{img_file}'), 
                       bbox_inches='tight')
            plt.close()
    
    # Сохранение результатов в JSON
    with open(os.path.join(output_dir, 'results.json'), 'w') as f:
        json.dump(results, f, indent=2)
    
    return results
```
